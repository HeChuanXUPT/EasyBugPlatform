-- 项目表
CREATE TABLE project (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    project_name VARCHAR(50) NOT NULL COMMENT '项目名称',
    create_time DATETIME DEFAULT NOW()
);

-- 用户表
CREATE TABLE sys_user (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(30) NOT NULL UNIQUE,
    password VARCHAR(60) NOT NULL,
    role VARCHAR(10) DEFAULT 'USER' COMMENT 'ADMIN/USER',
    create_time DATETIME DEFAULT NOW()
);

-- 问题表（核心）
CREATE TABLE issue (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    title VARCHAR(100) NOT NULL COMMENT '问题标题',
    project_id BIGINT NOT NULL COMMENT '所属项目',
    issue_type VARCHAR(20) COMMENT 'BUG/需求反馈/优化建议',
    severity VARCHAR(10) COMMENT 'P0/P1/P2/P3',
    priority VARCHAR(10) COMMENT '高/中/低',
    steps TEXT COMMENT '复现步骤',
    reporter_id BIGINT NOT NULL COMMENT '上报人ID',
    assignee_id BIGINT COMMENT '负责人ID',
    status VARCHAR(20) DEFAULT '待指派' COMMENT '状态',
    close_reason TEXT COMMENT '关闭原因',
    create_time DATETIME DEFAULT NOW(),
    update_time DATETIME DEFAULT NOW()
);

-- 问题流转记录表
CREATE TABLE issue_flow (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    issue_id BIGINT NOT NULL,
    operator_id BIGINT NOT NULL,
    old_status VARCHAR(20),
    new_status VARCHAR(20) NOT NULL,
    remark TEXT COMMENT '原因',
    create_time DATETIME DEFAULT NOW()
);

-- 评论表
CREATE TABLE issue_comment (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    issue_id BIGINT NOT NULL,
    user_id BIGINT NOT NULL,
    content TEXT NOT NULL,
    create_time DATETIME DEFAULT NOW()
);
