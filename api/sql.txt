-- CREATE DATABASE bug_platform;

-- 项目表
CREATE TABLE project (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    project_name VARCHAR(50) NOT NULL COMMENT '项目名称',
    create_time DATETIME DEFAULT NOW()
);

-- 用户表
CREATE TABLE sys_user (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(30) NOT NULL UNIQUE,
    password VARCHAR(60) NOT NULL,
    role VARCHAR(10) DEFAULT 'USER' COMMENT 'ADMIN/USER',
    create_time DATETIME DEFAULT NOW()
);

-- 问题表（核心）
CREATE TABLE `issue` (
	`id` BIGINT NOT NULL AUTO_INCREMENT,
	`title` VARCHAR(100) NOT NULL COMMENT '问题标题' COLLATE 'utf8mb4_0900_ai_ci',
	`project_id` BIGINT NOT NULL COMMENT '所属项目',
	`module` VARCHAR(50) NULL DEFAULT NULL COMMENT '模块' COLLATE 'utf8mb4_0900_ai_ci',
	`issue_type` VARCHAR(20) NULL DEFAULT NULL COMMENT 'BUG/需求反馈/优化建议' COLLATE 'utf8mb4_0900_ai_ci',
	`severity` VARCHAR(10) NULL DEFAULT NULL COMMENT 'P0/P1/P2/P3' COLLATE 'utf8mb4_0900_ai_ci',
	`priority` VARCHAR(10) NULL DEFAULT NULL COMMENT '高/中/低' COLLATE 'utf8mb4_0900_ai_ci',
	`content` TEXT NULL DEFAULT NULL COMMENT '问题描述' COLLATE 'utf8mb4_0900_ai_ci',
	`reporter_id` BIGINT NOT NULL COMMENT '上报人ID',
	`assignee_id` BIGINT NULL DEFAULT NULL COMMENT '负责人ID',
	`status` VARCHAR(20) NULL DEFAULT '待指派' COMMENT '状态' COLLATE 'utf8mb4_0900_ai_ci',
	`solution` TEXT NULL DEFAULT NULL COMMENT '解决方法' COLLATE 'utf8mb4_0900_ai_ci',
	`create_time` DATETIME NULL DEFAULT (CURRENT_TIMESTAMP),
	`update_time` DATETIME NULL DEFAULT (CURRENT_TIMESTAMP),
	PRIMARY KEY (`id`) USING BTREE
)
;


-- 问题流转记录表
CREATE TABLE issue_flow (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    issue_id BIGINT NOT NULL,
    operator_id BIGINT NOT NULL,
    old_status VARCHAR(20),
    new_status VARCHAR(20) NOT NULL,
    remark TEXT COMMENT '原因',
    create_time DATETIME DEFAULT NOW()
);

-- 评论表
CREATE TABLE issue_comment (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    issue_id BIGINT NOT NULL,
    user_id BIGINT NOT NULL,
    content TEXT NOT NULL,
    create_time DATETIME DEFAULT NOW()
);
