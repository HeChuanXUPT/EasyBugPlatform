-- bug_platform 的数据库结构
CREATE DATABASE IF NOT EXISTS `bug_platform` ;
USE `bug_platform`;

-- 表 bug_platform.issue 结构
CREATE TABLE IF NOT EXISTS `issue` (
  `id` bigint NOT NULL AUTO_INCREMENT,
  `title` varchar(100) NOT NULL COMMENT '问题标题',
  `project_id` bigint NOT NULL COMMENT '所属项目',
  `module` varchar(50) DEFAULT NULL COMMENT '模块',
  `issue_type` varchar(20) DEFAULT NULL COMMENT 'BUG/需求反馈/优化建议',
  `severity` varchar(10) DEFAULT NULL COMMENT 'P0/P1/P2/P3',
  `priority` varchar(10) DEFAULT NULL COMMENT '高/中/低',
  `content` text CHARACTER SET utf8mb4 COMMENT '问题描述',
  `reporter_id` bigint NOT NULL COMMENT '上报人ID',
  `assignee_id` bigint DEFAULT NULL COMMENT '负责人ID',
  `status` varchar(20) CHARACTER SET utf8mb4 DEFAULT '处理中' COMMENT '状态：处理中、待验证、已验证、关闭',
  `solution` text CHARACTER SET utf8mb4 COMMENT '解决方法',
  `create_time` datetime DEFAULT CURRENT_TIMESTAMP,
  `update_time` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`)
);

--  表 bug_platform.issue_comment 结构
CREATE TABLE IF NOT EXISTS `issue_comment` (
  `id` bigint NOT NULL AUTO_INCREMENT,
  `issue_id` bigint NOT NULL,
  `user_id` bigint NOT NULL,
  `content` text NOT NULL,
  `create_time` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`)
);


--  表 bug_platform.issue_flow 结构
CREATE TABLE IF NOT EXISTS `issue_flow` (
  `id` bigint NOT NULL AUTO_INCREMENT,
  `issue_id` bigint NOT NULL,
  `operator_id` bigint NOT NULL,
  `old_status` varchar(20) DEFAULT NULL,
  `new_status` varchar(20) NOT NULL,
  `remark` text COMMENT '原因',
  `create_time` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`)
);

--  表 bug_platform.project 结构
CREATE TABLE IF NOT EXISTS `project` (
  `id` bigint NOT NULL AUTO_INCREMENT,
  `project_name` varchar(50) NOT NULL COMMENT '项目名称',
  `create_time` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`)
);

-- 表 bug_platform.sys_user 结构
CREATE TABLE IF NOT EXISTS `sys_user` (
  `id` bigint NOT NULL AUTO_INCREMENT,
  `username` varchar(30) NOT NULL,
  `password` varchar(60) NOT NULL,
  `role` varchar(10) DEFAULT 'USER' COMMENT 'ADMIN/USER',
  `create_time` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `username` (`username`)
);
